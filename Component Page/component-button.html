/* primary button variants */
.primary-button {
  background-color: var(--button-primary-color);
  color: var(--text-secondary-color-hover);
  padding: var(--spacing-small) var(--spacing-small);
  border: none;
  font-size: var(--text-description);
  font-weight: var(--font-weight-medium);
  cursor: pointer;
}
.primary-button:hover {
  background-color: var(--button-primary-color-hover);
  color: var(--text-secondary-color-hover);
  transition: all 0.3s ease;
}
.primary-button:active {
  background-color: var(--button-primary-color-select);
  color: var(--text-highlight-color);
  transition: all 0.1s ease;
}

/* secondary button variants */
.secondary-button {
  background-color: var(--button-secondary-color);
  color: var(--text-secondary-color-hover);
  padding: var(--spacing-small) var(--spacing-small);
  border: none;
  font-size: var(--text-description);
  font-weight: var(--font-weight-medium);
  cursor: pointer;
}
.secondary-button:hover {
  background-color: var(--button-secondary-color-hover);
  color: var(--text-primary-color);
  transition: all 0.3s ease;
}
.secondary-button:active {
  background-color: var(--button-primary-color-select);
  color: var(--text-secondary-color);
  transition: all 0.1s ease;
}

/* accent button variants */
.accent-button {
  background-color: var(--button-primary-color-hover);
  color: var(--text-secondary-color-hover);
  padding: var(--spacing-small) var(--spacing-small);
  border: none;
  font-size: var(--text-description);
  font-weight: var(--font-weight-medium);
  cursor: pointer;
}
.accent-button:hover {
  background-color: var(--button-secondary-color-hover);
  color: var(--text-primary-color-hover);
  transition: all 0.3s ease;
}
.accent-button:active {
  background-color: var(--button-primary-color-select);
  color: var(--text-highlight-color);
  transition: all 0.1s ease;
}

/* disabled button variant (shared between all buttons) */
.button-disabled {
  background-color: var(--button-secondary-color);
  color: var(--text-primary-color);
  padding: var(--spacing-small) var(--spacing-small);
  border: none;
  font-size: var(--text-description);
  font-weight: var(--font-weight-medium);
  cursor: disabled;
